#   Selection Sort works like:
    scan all your cards, pick the smallest one, put it first. Then scan the rest, pick the next smallest,
    put it second. Repeat until done

#   Code Breakdown in C++

    #include <bits/stdc++.h>
    using namespace std;

    void selectionSort(vector<int> &arr) {   // pass by reference — modifies in-place
        int n = arr.size();

        for (int i = 0; i < n - 1; ++i) {   // outer loop: boundary of sorted portion
                                            // stops at n-2; last element falls into place
            int min_idx = i;                  // assume current position is the minimum

            for (int j = i + 1; j < n; ++j) {  // inner loop: scan unsorted portion
                if (arr[j] < arr[min_idx]) {
                    min_idx = j;             // update index if smaller found (not value!)
                }
            }
            swap(arr[i], arr[min_idx]);      // STL swap — single swap per pass (key advantage)
        }
    }

#   Code Breakdown in Java

    static void selectionSort(int[] arr) {
        int n = arr.length;                 // array is passed by reference in Java

        for (int i = 0; i < n - 1; i++) {
            int min_idx = i;

            for (int j = i + 1; j < n; j++) {
                if (arr[j] < arr[min_idx]) {
                    min_idx = j;
                }
            }
            // Manual swap — Java has no built-in swap for primitives
            int temp = arr[i];
            arr[i] = arr[min_idx];
            arr[min_idx] = temp;
        }
    }

#   Trade-off

    C++ specific:
    - Forgetting & in vector<int> &arr — without it, you're sorting a copy, not the original

    Java specific:
    - Calling arr.length() with parentheses — it's arr.length (a field, not a method). Classic interview trap.
